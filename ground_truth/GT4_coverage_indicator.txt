{ ============================================================================
  GT4 - Extended coverage ground-truth  (MultiCharts PowerLanguage)
  ----------------------------------------------------------------------------
  TYPE:  INDICATOR  (PowerLanguage Editor -> New -> Indicator)

  PURPOSE
    Pushes front-end / function coverage further than GT3: date/time functions,
    daily-session data, volume/tick reserved words, single-return regression &
    trend indicators (ADX/ADXR/DMI/Bollinger/LinearReg*/Correlation), the
    MULTI-OUTPUT var-parameter call convention (Stochastic), plus new syntax:
    `//` line comments, `Repeat..Until`, an `IntrabarPersist` variable, `Plot2`,
    `NoPlot`, the `true` literal, and a `Condition` boolean var.

    Indicator (no orders) so nothing conflicts. One value per line; flat CSV row.

  GROUNDING
    Every token grounded in the local refs (PowerLanguage Keyword Reference +
    TradeStation EL reference); signatures from the TS reference. See GROUNDING.md.

  HIGHEST COMPILE-RISK LINES (if the Editor rejects GT4, these are the first
  suspects to isolate into a separate file - per the agreed "try it, then split"):
    - the Stochastic(...) 11-arg var-parameter call (oFastK..oSlowD outputs)
    - the Repeat .. Until block
  Because each value is computed on its own line, a single rejected line can be
  commented out (set its column to 0) without affecting the rest of the capture.

  RUN
    Delete any old OutPath file first (Print(File()) appends). Apply to a fixed
    symbol / interval / date-range and record them. Reload once to capture.
============================================================================ }

Inputs:
	Len( 14 ),
	Prec( 6 ),
	OutPath( "C:\pl_gt\GT4_coverage.csv" ) ;

{ IntrabarPersist variable - declared in its own block to scope the modifier }
Vars:
	IntrabarPersist IbpCount( 0 ) ;

Vars:
	AlwaysOn( true ),
	Tan45( 0 ),
	TypPrice( 0 ),
	WClose( 0 ),
	LRSlope( 0 ),
	LRAngle( 0 ),
	Correl( 0 ),
	AdxV( 0 ),
	AdxrV( 0 ),
	DmiP( 0 ),
	DmiM( 0 ),
	BbUp( 0 ),
	BbDn( 0 ),
	oFastK( 0 ),
	oFastD( 0 ),
	oSlowK( 0 ),
	oSlowD( 0 ),
	StochRet( 0 ),
	ClD( 0 ),
	OpD( 0 ),
	HiD( 0 ),
	LoD( 0 ),
	MnV( 0 ),
	YrV( 0 ),
	DomV( 0 ),
	CalcD( 0 ),
	CalcT( 0 ),
	CurDate( 0 ),
	CurTime( 0 ),
	TicksV( 0 ),
	UpT( 0 ),
	DnT( 0 ),
	OiV( 0 ),
	Cond1N( 0 ),
	AlwaysOnN( 0 ),
	RepeatOut( 0 ),
	Line( "" ) ;

// this is a // line comment - exercises the second EL comment style
Tan45 = Tangent( 45 ) ;                 { degrees -> 1.0 (constant probe) }

{ ---- price/session functions ---- }
TypPrice = TypicalPrice ;
WClose = WeightedClose ;

{ ---- single-return regression / trend indicators ---- }
LRSlope = LinearRegSlope( Close, Len ) ;
LRAngle = LinearRegAngle( Close, Len ) ;
Correl = Correlation( Close, Open, Len ) ;
AdxV = ADX( Len ) ;
AdxrV = ADXR( Len ) ;
DmiP = DMIPlus( Len ) ;
DmiM = DMIMinus( Len ) ;
BbUp = BollingerBand( Close, Len, 2 ) ;
BbDn = BollingerBand( Close, Len, -2 ) ;

{ ---- multi-output (var-parameter) call: Stochastic sets oFastK..oSlowD ---- }
StochRet = Stochastic( High, Low, Close, Len, 3, 3, 1, oFastK, oFastD, oSlowK, oSlowD ) ;

{ ---- daily-session data ---- }
ClD = CloseD( 0 ) ;
OpD = OpenD( 0 ) ;
HiD = HighD( 0 ) ;
LoD = LowD( 0 ) ;

{ ---- date / time functions ---- }
MnV = Month( Date ) ;
YrV = Year( Date ) ;
DomV = DayOfMonth( Date ) ;
CalcD = CalcDate( Date, -14 ) ;
CalcT = CalcTime( Time, -15 ) ;
CurDate = CurrentDate ;
CurTime = CurrentTime ;

{ ---- volume / tick reserved words ---- }
TicksV = Ticks ;
UpT = UpTicks ;
DnT = DownTicks ;
OiV = OpenInt ;

{ ---- IntrabarPersist counter ---- }
IbpCount = IbpCount + 1 ;

{ ---- Condition var + true literal ---- }
Condition1 = Close > Open ;
Cond1N = IFF( Condition1, 1, 0 ) ;
AlwaysOnN = IFF( AlwaysOn, 1, 0 ) ;     { true literal -> always 1 }

{ ---- Repeat .. Until loop - constant probe: counts to 5 ---- }
RepeatOut = 0 ;
Repeat
	RepeatOut = RepeatOut + 1 ;
Until RepeatOut >= 5 ;

{ ---- header row, written once ---- }
Once Print( File( OutPath ), "bn,date,time,open,high,low,close,volume,tan45,typprice,wclose,lrslope,lrangle,correl,adx,adxr,dmiplus,dmiminus,bbup,bbdn,stochret,stochk,stochd,closed,opend,highd,lowd,month,year,dom,calcdate,calctime,curdate,curtime,ticks,upticks,downticks,openint,ibpcount,cond1,alwayson,repeatout" ) ;

{ ---- build the CSV row: flat concatenation, no nesting ---- }
Line = NumToStr( BarNumber, 0 ) + "," +
	NumToStr( Date, 0 ) + "," +
	NumToStr( Time, 0 ) + "," +
	NumToStr( Open, Prec ) + "," +
	NumToStr( High, Prec ) + "," +
	NumToStr( Low, Prec ) + "," +
	NumToStr( Close, Prec ) + "," +
	NumToStr( Volume, 0 ) + "," +
	NumToStr( Tan45, Prec ) + "," +
	NumToStr( TypPrice, Prec ) + "," +
	NumToStr( WClose, Prec ) + "," +
	NumToStr( LRSlope, Prec ) + "," +
	NumToStr( LRAngle, Prec ) + "," +
	NumToStr( Correl, Prec ) + "," +
	NumToStr( AdxV, Prec ) + "," +
	NumToStr( AdxrV, Prec ) + "," +
	NumToStr( DmiP, Prec ) + "," +
	NumToStr( DmiM, Prec ) + "," +
	NumToStr( BbUp, Prec ) + "," +
	NumToStr( BbDn, Prec ) + "," +
	NumToStr( StochRet, 0 ) + "," +
	NumToStr( oSlowK, Prec ) + "," +
	NumToStr( oSlowD, Prec ) + "," +
	NumToStr( ClD, Prec ) + "," +
	NumToStr( OpD, Prec ) + "," +
	NumToStr( HiD, Prec ) + "," +
	NumToStr( LoD, Prec ) + "," +
	NumToStr( MnV, 0 ) + "," +
	NumToStr( YrV, 0 ) + "," +
	NumToStr( DomV, 0 ) + "," +
	NumToStr( CalcD, 0 ) + "," +
	NumToStr( CalcT, 0 ) + "," +
	NumToStr( CurDate, 0 ) + "," +
	NumToStr( CurTime, 0 ) + "," +
	NumToStr( TicksV, 0 ) + "," +
	NumToStr( UpT, 0 ) + "," +
	NumToStr( DnT, 0 ) + "," +
	NumToStr( OiV, 0 ) + "," +
	NumToStr( IbpCount, 0 ) + "," +
	NumToStr( Cond1N, 0 ) + "," +
	NumToStr( AlwaysOnN, 0 ) + "," +
	NumToStr( RepeatOut, 0 ) ;

Print( File( OutPath ), Line ) ;

{ plots: Plot2 and NoPlot exercise additional plotting front-end }
Plot1( TypPrice ) ;
Plot2( WClose ) ;
If BarNumber < 0 Then
	NoPlot( 1 ) ;
